135template<class EigenVector_, typename Index_, typename Block_>
136std::optional<BlockingDetails<EigenVector_> > compute_blocking_details(
137 const Index_ ncells,
138 const Block_* block,
139 const std::size_t num_blocks,
140 const std::vector<Index_>& block_sizes,
141 const scran_blocks::WeightPolicy block_weight_policy,
142 const scran_blocks::VariableWeightParameters& variable_block_weight_parameters
143) {
144 if (block_weight_policy == scran_blocks::WeightPolicy::NONE) {
145 return std::optional<BlockingDetails<EigenVector_> >();
146 }
147
148 BlockingDetails<EigenVector_> output(num_blocks, ncells);
149 auto& total_weight = output.total_block_weight;
150 auto& element_weight = output.per_element_weight;
151
152 for (std::size_t b = 0; b < num_blocks; ++b) {
153 const auto bsize = block_sizes[b];
154
155 // Computing effective block weights that also incorporate division by the
156 // block size. This avoids having to do the division by block size in the
157 // 'compute_blockwise_mean_and_variance*()' functions.
158 if (bsize) {
159 typename EigenVector_::Scalar block_weight = 1;
160 if (block_weight_policy == scran_blocks::WeightPolicy::VARIABLE) {
161 block_weight = scran_blocks::compute_variable_weight(bsize, variable_block_weight_parameters);
162 }
163
164 element_weight[b] = block_weight / bsize;
165 total_weight += block_weight;
166 } else {
167 element_weight[b] = 0;
168 }
169 }
170
171 // Setting a placeholder value to avoid problems with division by zero.
172 if (total_weight == 0) {
173 total_weight = 1;
174 }
175
176 // Expanding them for multiplication in the IRLBA wrappers.
177 auto sqrt_weights = element_weight;
178 for (auto& s : sqrt_weights) {
179 s = std::sqrt(s);
180 }
181
182 auto& expanded = output.expanded_weights;
183 for (Index_ c = 0; c < ncells; ++c) {
184 expanded.coeffRef(c) = sqrt_weights[block[c]];
185 }
186
187 return output;
188}

194template<class IrlbaSparseMatrix_, typename Block_, class Index_, class EigenVector_, class EigenMatrix_>
195void compute_blockwise_mean_and_variance_realized_sparse(
196 const IrlbaSparseMatrix_& emat, // this should be column-major with genes in the columns.
197 const Block_* block,
198 const std::size_t num_blocks,
199 const std::vector<Index_>& block_sizes,
200 const std::optional<BlockingDetails<EigenVector_> >& block_details,
201 EigenMatrix_& centers,
202 EigenVector_& variances,
203 const int nthreads
204) {
205 const auto ngenes = emat.cols();
206 const auto ncells = emat.rows();
207 const auto& values = emat.get_values();
208 const auto& indices = emat.get_indices();
209 const auto& pointers = emat.get_pointers();
210 static_assert(!EigenMatrix_::IsRowMajor);
211
212 assert(sanisizer::is_equal(ngenes, variances.size()));
213 assert(sanisizer::is_equal(ngenes, centers.cols()));
214 assert(sanisizer::is_equal(num_blocks, centers.rows()));
215
216 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
217 auto block_zeros = sanisizer::create<std::vector<Index_> >(num_blocks);
218 auto block_rss = sanisizer::create<std::vector<typename EigenVector_::Scalar> >(num_blocks);
219 auto block_centers = sanisizer::create<std::vector<typename EigenMatrix_::Scalar> >(num_blocks); // use a local copy to avoid false sharing.
220
221 for (I<decltype(start)> g = start, end = start + length; g < end; ++g) {
222 const auto offset = pointers[g];
223 const auto num_nonzero = pointers[g + 1] - offset; // increment won't overflow as 'g < end' and 'end' is of the same type.
224
225 const auto vptr = values.data() + offset;
226 const auto iptr = indices.data() + offset;
227
228 std::fill(block_centers.begin(), block_centers.end(), 0);
229 for (I<decltype(num_nonzero)> i = 0; i < num_nonzero; ++i) {
230 block_centers[block[iptr[i]]] += vptr[i];
231 }
232 for (std::size_t b = 0; b < num_blocks; ++b) {
233 const auto bsize = block_sizes[b];
234 if (bsize) {
235 block_centers[b] /= bsize;
236 }
237 // We'll set the means to NaN at the end of the function.
238 // We don't do it here as NaNs would propagate in ResidualMatrix's multiply.
239 }
240
241 // Computing the RSS instead of the sample variance.
242 // We don't consider the loss of residual d.f. from estimating the block means, as the PCA doesn't either.
243 std::copy(block_sizes.begin(), block_sizes.end(), block_zeros.begin());
244 std::fill(block_rss.begin(), block_rss.end(), 0);
245
246 for (I<decltype(num_nonzero)> i = 0; i < num_nonzero; ++i) {
247 const Block_ curb = block[iptr[i]];
248 const auto diff = vptr[i] - block_centers[curb];
249 block_rss[curb] += diff * diff;
250 --block_zeros[curb];
251 }
252
253 typename EigenVector_::Scalar rss = 0;
254 for (std::size_t b = 0; b < num_blocks; ++b) {
255 const auto bsize = block_sizes[b];
256 if (bsize) {
257 const auto val = block_centers[b];
258 const auto final_rss = block_rss[b] + val * val * block_zeros[b];
259 if (block_details.has_value()) {
260 rss += final_rss * block_details->per_element_weight[b];
261 } else {
262 rss += final_rss;
263 }
264 }
265 }
266
267 // COMMENT ON DENOMINATOR:
268 // If we're not dealing with weights, we compute the actual sample variance for easy interpretation
269 // (and to match up with the per-PC calculations in clean_up).
270 //
271 // If we're dealing with weights, the concept of the sample variance becomes somewhat weird.
272 // So, we just use the same denominator for consistency in clean_up_projected.
273 // Magnitude doesn't matter when scaling for process_scale_vector anyway.
274 //
275 // If there are not enough cells, we set the variance to zero so that no scaling is done in process_scale_vector().
276 // We'll set this to NaN at the end of the function.
277 if (ncells > 1) {
278 variances[g] = rss / (ncells - 1);
279 } else {
280 variances[g] = 0;
281 }
282
283 std::copy(block_centers.begin(), block_centers.end(), centers.data() + sanisizer::product_unsafe<std::size_t>(g, num_blocks));
284 }
285 }, ngenes, nthreads);
286}

356template<typename Value_, typename Index_, typename Block_, class EigenMatrix_, class EigenVector_>
357void compute_blockwise_mean_and_variance_tatami(
358 const tatami::Matrix<Value_, Index_>& mat, // this should have genes in the rows!
359 const Block_* block,
360 const std::size_t num_blocks,
361 const std::vector<Index_>& block_sizes,
362 const std::optional<BlockingDetails<EigenVector_> >& block_details,
363 EigenMatrix_& centers,
364 EigenVector_& variances,
365 const int nthreads
366) {
367 static_assert(!EigenMatrix_::IsRowMajor); // need this for correct pointer calculations.
368 typedef typename EigenMatrix_::Scalar Float;
369
370 const auto ngenes = mat.nrow();
371 EigenMatrix_ tmp_mean(
372 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().rows())> >(ngenes),
373 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().cols())> >(num_blocks)
374 );
375
376 tatami_stats::GroupRssBuffers<Float> buffers;
377 buffers.mean.reserve(num_blocks);
378 buffers.rss.reserve(num_blocks);
379 auto tmp_rss = sanisizer::create<std::vector<std::vector<Float> > >(num_blocks);
380
381 for (std::size_t b = 0; b < num_blocks; ++b) {
382 buffers.mean.push_back(tmp_mean.data() + sanisizer::product_unsafe<std::size_t>(ngenes, b));
383 tatami::resize_container_to_Index_size(tmp_rss[b], ngenes);
384 buffers.rss.push_back(tmp_rss[b].data());
385 }
386
387 tatami_stats::GroupRssOptions opt;
388 opt.num_threads = nthreads;
389 tatami_stats::group_rss(true, mat, block, num_blocks, block_sizes.data(), buffers, opt);
390
391 assert(sanisizer::is_equal(variances.size(), ngenes));
392 variances.setZero();
393 for (std::size_t b = 0; b < num_blocks; ++b) {
394 if (block_sizes[b]) {
395 const auto& currss = tmp_rss[b];
396 if (block_details.has_value()) {
397 for (Index_ g = 0; g < ngenes; ++g) {
398 variances.coeffRef(g) += currss[g] * block_details->per_element_weight[b];
399 }
400 } else {
401 for (Index_ g = 0; g < ngenes; ++g) {
402 variances.coeffRef(g) += currss[g];
403 }
404 }
405 } else {
406 // Replace NaNs with zeros so ResidualMatrix's multiplications don't propagate NaNs.
407 std::fill_n(buffers.mean[b], ngenes, 0);
408 }
409 }
410
411 centers = tmp_mean.adjoint();
412
413 // See COMMENT ON DENOMINATOR above.
414 const auto ncells = mat.ncol();
415 if (ncells > 1) {
416 for (Index_ g = 0; g < ngenes; ++g) {
417 variances.coeffRef(g) /= ncells - 1;
418 }
419 }
420}

436template<class EigenVector_, class IrlbaSparseMatrix_, class EigenMatrix_>
437inline void project_matrix_realized_sparse(
438 const IrlbaSparseMatrix_& emat, // cell in rows, genes in the columns, CSC.
439 EigenMatrix_& components, // dims in rows, cells in columns
440 const EigenMatrix_& scaled_rotation, // genes in rows, dims in columns
441 int nthreads
442) {
443 const auto rank = scaled_rotation.cols();
444 const auto ncells = emat.rows();
445 const auto ngenes = emat.cols();
446
447 // Store as transposed for more cache efficiency.
448 components.resize(
449 sanisizer::cast<I<decltype(components.rows())> >(rank),
450 sanisizer::cast<I<decltype(components.cols())> >(ncells)
451 );
452 components.setZero();
453
454 const auto& values = emat.get_values();
455 const auto& indices = emat.get_indices();
456 const auto& pointers = emat.get_pointers();
457
458 if (nthreads == 1) {
459 auto multipliers = sanisizer::create<EigenVector_>(rank);
460 for (I<decltype(ngenes)> g = 0; g < ngenes; ++g) {
461 multipliers.noalias() = scaled_rotation.row(g);
462 const auto start = pointers[g], end = pointers[g + 1]; // increment is safe as 'g + 1 <= ngenes'.
463 for (auto i = start; i < end; ++i) {
464 components.col(indices[i]).noalias() += values[i] * multipliers;
465 }
466 }
467
468 } else {
469 // Here, the general strategy is to split the matrix by chunks into genes,
470 // perform the matrix multiplication for each chunk,
471 // and then sum the per-chunk products to obtain the final product.
472 // The exact result of the reduction depends on the number of threads,
473 // but this is an acceptable annoyance for greater speed.
474 const auto& primary_bounds = emat.get_primary_boundaries();
475 auto working = sanisizer::create<std::vector<EigenMatrix_> >(nthreads - 1);
476
477 irlba::parallelize(nthreads, [&](const int t) -> void {
478 EigenMatrix_* ptr;
479 if (t == 0) {
480 ptr = &components;
481 } else {
482 auto& mat = working[t - 1];
483 mat.resize(components.rows(), components.cols());
484 mat.setZero();
485 ptr = &mat;
486 }
487
488 const auto gstart = primary_bounds[t];
489 const auto gend = primary_bounds[t + 1]; // increment is safe as 't + 1 <= nthreads'.
490 auto multipliers = sanisizer::create<EigenVector_>(rank);
491 for (I<decltype(ngenes)> g = gstart; g < gend; ++g) {
492 multipliers.noalias() = scaled_rotation.row(g);
493 const auto start = pointers[g], end = pointers[g + 1]; // increment is safe as 'g + 1 <= ngenes'
494 for (auto i = start; i < end; ++i) {
495 ptr->col(indices[i]).noalias() += values[i] * multipliers;
496 }
497 }
498 });
499
500 for (auto& w : working) {
501 components += w;
502 }
503 }
504}

506template<typename Value_, typename Index_, class EigenMatrix_>
507void project_matrix_transposed_tatami(
508 const tatami::Matrix<Value_, Index_>& mat, // genes in rows, cells in columns
509 EigenMatrix_& components,
510 const EigenMatrix_& scaled_rotation, // genes in rows, dims in columns
511 const int nthreads)
512{
513 const auto rank = scaled_rotation.cols();
514 const auto ngenes = mat.nrow();
515 const auto ncells = mat.ncol();
516 typedef typename EigenMatrix_::Scalar Scalar;
517
518 // Store as transposed for more cache efficiency.
519 components.resize(
520 sanisizer::cast<I<decltype(components.rows())> >(rank),
521 sanisizer::cast<I<decltype(components.cols())> >(ncells)
522 );
523
524 if (mat.prefer_rows()) {
525 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
526 static_assert(!EigenMatrix_::IsRowMajor);
527 const auto vptr = scaled_rotation.data();
528 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(length);
529
530 std::vector<std::vector<Scalar> > local_buffers; // create separate buffers to avoid false sharing.
531 local_buffers.reserve(rank);
532 for (I<decltype(rank)> r = 0; r < rank; ++r) {
533 local_buffers.emplace_back(tatami::cast_Index_to_container_size<I<decltype(local_buffers.front())> >(length));
534 }
535
536 if (mat.is_sparse()) {
537 auto ibuffer = tatami::create_container_of_Index_size<std::vector<Index_> >(length);
538 auto ext = tatami::consecutive_extractor<true>(mat, true, static_cast<Index_>(0), ngenes, start, length);
539 for (Index_ g = 0; g < ngenes; ++g) {
540 const auto range = ext->fetch(vbuffer.data(), ibuffer.data());
541 for (I<decltype(rank)> r = 0; r < rank; ++r) {
542 const auto mult = vptr[sanisizer::nd_offset<std::size_t>(g, ngenes, r)];
543 auto& local_buffer = local_buffers[r];
544 for (Index_ i = 0; i < range.number; ++i) {
545 local_buffer[range.index[i] - start] += range.value[i] * mult;
546 }
547 }
548 }
549
550 } else {
551 auto ext = tatami::consecutive_extractor<false>(mat, true, static_cast<Index_>(0), ngenes, start, length);
552 for (Index_ g = 0; g < ngenes; ++g) {
553 const auto ptr = ext->fetch(vbuffer.data());
554 for (I<decltype(rank)> r = 0; r < rank; ++r) {
555 const auto mult = vptr[sanisizer::nd_offset<std::size_t>(g, ngenes, r)];
556 auto& local_buffer = local_buffers[r];
557 for (Index_ i = 0; i < length; ++i) {
558 local_buffer[i] += ptr[i] * mult;
559 }
560 }
561 }
562 }
563
564 for (I<decltype(rank)> r = 0; r < rank; ++r) {
565 for (Index_ c = 0; c < length; ++c) {
566 components.coeffRef(r, c + start) = local_buffers[r][c];
567 }
568 }
569
570 }, ncells, nthreads);
571
572 } else {
573 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
574 static_assert(!EigenMatrix_::IsRowMajor);
575 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(ngenes);
576
577 if (mat.is_sparse()) {
578 std::vector<Index_> ibuffer(ngenes);
579 auto ext = tatami::consecutive_extractor<true>(mat, false, start, length);
580
581 for (Index_ c = start, end = start + length; c < end; ++c) {
582 const auto range = ext->fetch(vbuffer.data(), ibuffer.data());
583 static_assert(!EigenMatrix_::IsRowMajor);
584 for (I<decltype(rank)> r = 0; r < rank; ++r) {
585 auto& output = components.coeffRef(r, c);
586 output = 0;
587 const auto rotptr = scaled_rotation.data() + sanisizer::product_unsafe<std::size_t>(r, ngenes);
588 for (Index_ i = 0; i < range.number; ++i) {
589 output += rotptr[range.index[i]] * range.value[i];
590 }
591 }
592 }
593
594 } else {
595 auto ext = tatami::consecutive_extractor<false>(mat, false, start, length);
596 for (Index_ c = start, end = start + length; c < end; ++c) {
597 const auto ptr = ext->fetch(vbuffer.data());
598 static_assert(!EigenMatrix_::IsRowMajor);
599 for (I<decltype(rank)> r = 0; r < rank; ++r) {
600 const auto rotptr = scaled_rotation.data() + sanisizer::product_unsafe<std::size_t>(r, ngenes);
601 components.coeffRef(r, c) = std::inner_product(rotptr, rotptr + ngenes, ptr, static_cast<Scalar>(0));
602 }
603 }
604 }
605 }, ncells, nthreads);
606 }

609template<class EigenMatrix_, class EigenVector_>
610void clean_up_projected(EigenMatrix_& projected, EigenVector_& D) {
611 // Empirically centering to give nice centered PCs, because we can't
612 // guarantee that the projection is centered in this manner.
613 for (I<decltype(projected.rows())> i = 0, prows = projected.rows(); i < prows; ++i) {
614 projected.row(i).array() -= projected.row(i).sum() / projected.cols();
615 }
616
617 // Just dividing by the number of observations - 1 regardless of weighting.
618 const typename EigenMatrix_::Scalar denom = projected.cols() - 1;
619 for (auto& d : D) {
620 d = d * d / denom;
621 }
622}

628template<class EigenVector_, class IrlbaMatrix_, typename Block_, class CenterMatrix_>
629class ResidualWorkspace final : public irlba::Workspace<EigenVector_> {
630public:
631 ResidualWorkspace(const IrlbaMatrix_& matrix, const Block_* block, const CenterMatrix_& means) :
632 my_work(matrix.new_known_workspace()),
633 my_block(block),
634 my_means(means),
635 my_sub(sanisizer::cast<I<decltype(my_sub.size())> >(my_means.rows()))
636 {}
637
638private:
639 I<decltype(std::declval<IrlbaMatrix_>().new_known_workspace())> my_work;
640 const Block_* my_block;
641 const CenterMatrix_& my_means;
642 EigenVector_ my_sub;
643
644public:
645 void multiply(const EigenVector_& right, EigenVector_& output) {
646 my_work->multiply(right, output);
647
648 my_sub.noalias() = my_means * right;
649 for (I<decltype(output.size())> i = 0, end = output.size(); i < end; ++i) {
650 auto& val = output.coeffRef(i);
651 val -= my_sub.coeff(my_block[i]);
652 }
653 }
654};

657class ResidualAdjointWorkspace final : public irlba::AdjointWorkspace<EigenVector_> {
658public:

672public:
673 void multiply(const EigenVector_& right, EigenVector_& output) {
674 my_work->multiply(right, output);
675
676 my_aggr.setZero();
677 for (I<decltype(right.size())> i = 0, end = right.size(); i < end; ++i) {
678 my_aggr.coeffRef(my_block[i]) += right.coeff(i);
679 }
680
681 output.noalias() -= my_means.adjoint() * my_aggr;
682 }

869 projector = [&](const EigenMatrix_& scaled_rotation) -> void {
870 project_matrix_transposed_tatami(mat, output.components, scaled_rotation, options.num_threads);
871 };

873 } else if (mat.sparse()) {

920 };

1006 /* column = */ false,

1014

1017 projector(scaled_rotation);
1018
1019 // Subtracting each block's mean from the PCs.
1020 if (options.center_scores_by_block) {
1021 EigenMatrix_ centering = (output.center * scaled_rotation).adjoint();
1022 for (I<decltype(ncells)> c =0 ; c < ncells; ++c) {
1023 output.components.col(c) -= centering.col(block[c]);
1024 }
1025 }
1026
1027 clean_up_projected(output.components, output.variance_explained);
1028 if (!options.transpose) {
1029 output.components.adjointInPlace();
1030 }
1031
1032 } else {
1033 output.metrics = irlba::compute(*ptr, options.number, output.components, output.rotation, output.variance_explained, options.irlba_options);
1034 subset_fun(num_blocks, block_sizes, block_details, output.components, output.variance_explained);
1035
1036 if (options.center_scores_by_block) {
1037 clean_up(mat.ncol(), output.components, output.variance_explained);
1038 if (options.transpose) {
1039 output.components.adjointInPlace();
1040 }
1041
1042 } else {
1043 EigenMatrix_ tmp;
1044 const auto& scaled_rotation = scale_rotation_matrix(output.rotation, options.scale, output.scale, tmp);
1045 projector(scaled_rotation);
1046
1047 clean_up_projected(output.components, output.variance_explained);
1048 if (!options.transpose) {
1049 output.components.adjointInPlace();
1050 }
1051 }
1052 }
1053
1054 if (!options.scale) {
1055 output.scale = EigenVector_();
1056 }

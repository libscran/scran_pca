90void compute_row_means_and_variances(const tatami::Matrix<Value_, Index_>& mat, const int num_threads, EigenVector_& center_v, EigenVector_& scale_v) {
91 tatami_stats::RssOptions vopt;
92 vopt.num_threads = num_threads;
93
94 tatami_stats::RssBuffers<typename EigenVector_::Scalar> buffers;
95 buffers.mean = center_v.data();
96 buffers.rss = scale_v.data();

98
99 const auto ncells = mat.ncol();
100 if (ncells > 1) {
101 scale_v /= ncells - 1;
102 } else if (!ncells) {

104 std::fill(center_v.begin(), center_v.end(), 0);

127template<class EigenMatrix_, typename Value_, typename Index_, class EigenVector_>
128std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > prepare_sparse_matrix_for_irlba(
129 const tatami::Matrix<Value_, Index_>& mat,
130 const SimplePcaOptions<EigenVector_>& options,
131 EigenVector_& center_v,
132 EigenVector_& scale_v,
133 typename EigenVector_::Scalar& total_var
134) {
135 const auto ngenes = mat.nrow();
136 sanisizer::resize(center_v, ngenes);
137 sanisizer::resize(scale_v, ngenes);
138 std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > output;
139
140 if (options.realize_matrix) {
141 // 'extracted' contains row-major contents...
142 auto extracted = tatami::retrieve_compressed_sparse_contents<Value_, Index_>(
143 mat,
144 /* row = */ true,
145 [&]{
146 tatami::RetrieveCompressedSparseContentsOptions opt;
147 opt.two_pass = false;
148 opt.num_threads = options.num_threads;
149 return opt;
150 }()
151 );
152
153 // But we effectively transpose it to CSC with genes in columns.
154 const Index_ ncells = mat.ncol();
155 const auto sparse_ptr = new irlba::ParallelSparseMatrix<
156 EigenVector_,
157 EigenMatrix_,
158 I<decltype(extracted.value)>,
159 I<decltype(extracted.index)>,
160 I<decltype(extracted.pointers)>
161 >(
162 ncells,
163 ngenes,
164 std::move(extracted.value),
165 std::move(extracted.index),
166 std::move(extracted.pointers),
167 true,
168 options.num_threads
169 );
170 output.reset(sparse_ptr);
171
172 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
173 const auto& pointers = sparse_ptr->get_pointers();
174 const auto& values = sparse_ptr->get_values();
175 quickstats::RssWorkspace<typename EigenVector_::Scalar> work;
176
177 for (Index_ g = start, end = start + length; g < end; ++g) {
178 const auto offset = pointers[g];
179 const auto next_offset = pointers[g + 1]; // increment won't overflow as 'g + 1 <= end'.
180 const Index_ num_nonzero = next_offset - offset;
181 const auto results = quickstats::rss(ncells, num_nonzero, values.data() + offset, work);
182 center_v.coeffRef(g) = results.mean;
183 scale_v.coeffRef(g) = results.rss;
184 }
185 }, ngenes, options.num_threads);
186
187 if (ncells > 1) {
188 // if there are fewer than 2 cells, scale_v will naturally be set to zero.
189 scale_v /= ncells - 1;
190 } else if (!ncells) {
191 // override quickstats::rss()'s setting of the mean to NaN if there are no cells.
192 std::fill(center_v.begin(), center_v.end(), 0);
193 }
194
195 total_var = process_scale_vector(options.scale, scale_v);
196
197 } else {
198 compute_row_means_and_variances(mat, options.num_threads, center_v, scale_v);
199 total_var = process_scale_vector(options.scale, scale_v);
200 output.reset(
201 new irlba_tatami::Transposed<EigenVector_, EigenMatrix_, Value_, Index_, decltype(&mat)>(&mat, options.num_threads)
202 );
203 }
204
205 return prepare_deferred_matrix_for_irlba(std::move(output), options, center_v, scale_v);
206}

208template<class EigenMatrix_, typename Value_, typename Index_, class EigenVector_>
209std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > prepare_dense_matrix_for_irlba(
210 const tatami::Matrix<Value_, Index_>& mat,
211 const SimplePcaOptions<EigenVector_>& options,
212 EigenVector_& center_v,
213 EigenVector_& scale_v,
214 typename EigenVector_::Scalar& total_var
215) {
216 const Index_ ngenes = mat.nrow();
217 sanisizer::resize(center_v, ngenes);
218 sanisizer::resize(scale_v, ngenes);
219
220 if (options.realize_matrix) {
221 // Create a matrix with genes in columns.
222 const Index_ ncells = mat.ncol();
223 auto emat = std::make_unique<EigenMatrix_>(
224 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().rows())> >(ncells),
225 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().cols())> >(ngenes)
226 );
227
228 // By default, Eigen's matrices are column major. In such cases, because we want to do
229 // a transposition, we pretend it's row major during the conversion.
230 static_assert(!EigenMatrix_::IsRowMajor);
231 tatami::convert_to_dense(
232 mat,
233 /* row_major = */ true,
234 emat->data(),
235 [&]{
236 tatami::ConvertToDenseOptions opt;
237 opt.num_threads = options.num_threads;
238 return opt;
239 }()
240 );
241
242 center_v.array() = emat->array().colwise().sum();
243 if (ncells) { // if there are no cells, center_v will naturally be set to zero.
244 center_v /= ncells;
245 }
246 emat->array().rowwise() -= center_v.adjoint().array(); // applying it to avoid wasting time with deferred operations inside IRLBA.
247
248 scale_v.array() = emat->array().colwise().squaredNorm();
249 if (ncells > 1) { // if there are fewer than 2 cells, scale_v will naturally be set to zero.
250 scale_v /= ncells - 1;
251 }
252
253 total_var = process_scale_vector(options.scale, scale_v);
254 if (options.scale) {
255 emat->array().rowwise() /= scale_v.adjoint().array();
256 }
257
258 return std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> >(
259 new irlba::SimpleMatrix<EigenVector_, EigenMatrix_, decltype(emat)>(std::move(emat))
260 );
261
262 } else {
263 compute_row_means_and_variances(mat, options.num_threads, center_v, scale_v);
264 total_var = process_scale_vector(options.scale, scale_v);
265 std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > output(
266 new irlba_tatami::Transposed<EigenVector_, EigenMatrix_, Value_, Index_, decltype(&mat)>(&mat, options.num_threads)
267 );
268 return prepare_deferred_matrix_for_irlba(std::move(output), options, center_v, scale_v);
269 }

85template<bool sparse_, typename Value_, typename Index_, class EigenVector_>
86void compute_row_means_and_variances(const tatami::Matrix<Value_, Index_>& mat, const int num_threads, EigenVector_& center_v, EigenVector_& scale_v) {
87 const auto ngenes = mat.nrow();
88
89 if (mat.prefer_rows()) {
90 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
91 auto ext = tatami::consecutive_extractor<sparse_>(mat, true, start, length, [&]{
92 tatami::Options opt;
93 opt.sparse_extract_index = false;
94 return opt;
95 }());
96 const auto ncells = mat.ncol();
97 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(ncells);
98
99 for (Index_ g = start, end = start + length; g < end; ++g) {
100 const auto results = [&]{
101 if constexpr(sparse_) {
102 auto range = ext->fetch(vbuffer.data(), NULL);
103 return tatami_stats::variances::direct(range.value, range.number, ncells, /* skip_nan = */ false);
104 } else {
105 auto ptr = ext->fetch(vbuffer.data());
106 return tatami_stats::variances::direct(ptr, ncells, /* skip_nan = */ false);
107 }
108 }();
109 center_v.coeffRef(g) = results.first;
110 scale_v.coeffRef(g) = results.second;
111 }
112 }, ngenes, num_threads);
113
114 } else {
115 tatami::parallelize([&](int t, Index_ start, Index_ length) -> void {
116 const auto ncells = mat.ncol();
117 auto ext = tatami::consecutive_extractor<sparse_>(mat, false, static_cast<Index_>(0), ncells, start, length);
118
119 typedef typename EigenVector_::Scalar Scalar;
120 tatami_stats::LocalOutputBuffer<Scalar> cbuffer(t, start, length, center_v.data());
121 tatami_stats::LocalOutputBuffer<Scalar> sbuffer(t, start, length, scale_v.data());
122
123 auto running = [&]{
124 if constexpr(sparse_) {
125 return tatami_stats::variances::RunningSparse<Scalar, Value_, Index_>(length, cbuffer.data(), sbuffer.data(), /* skip_nan = */ false, /* subtract = */ start);
126 } else {
127 return tatami_stats::variances::RunningDense<Scalar, Value_, Index_>(length, cbuffer.data(), sbuffer.data(), /* skip_nan = */ false);
128 }
129 }();
130
131 auto vbuffer = tatami::create_container_of_Index_size<std::vector<Value_> >(length);
132 auto ibuffer = [&]{
133 if constexpr(sparse_) {
134 return tatami::create_container_of_Index_size<std::vector<Index_> >(length);
135 } else {
136 return false;
137 }
138 }();
139
140 for (Index_ c = 0; c < ncells; ++c) {
141 if constexpr(sparse_) {
142 const auto range = ext->fetch(vbuffer.data(), ibuffer.data());
143 running.add(range.value, range.index, range.number);
144 } else {
145 const auto ptr = ext->fetch(vbuffer.data());
146 running.add(ptr);
147 }
148 }
149
150 running.finish();
151 cbuffer.transfer();
152 sbuffer.transfer();
153 }, ngenes, num_threads);
154 }
155}

176template<class EigenMatrix_, typename Value_, typename Index_, class EigenVector_>
177std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > prepare_sparse_matrix_for_irlba(
178 const tatami::Matrix<Value_, Index_>& mat,
179 const SimplePcaOptions<EigenVector_>& options,
180 EigenVector_& center_v,
181 EigenVector_& scale_v,
182 typename EigenVector_::Scalar& total_var
183) {
184 const auto ngenes = mat.nrow();
185 sanisizer::resize(center_v, ngenes);
186 sanisizer::resize(scale_v, ngenes);
187 std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > output;
188
189 if (options.realize_matrix) {
190 // 'extracted' contains row-major contents...
191 auto extracted = tatami::retrieve_compressed_sparse_contents<Value_, Index_>(
192 mat,
193 /* row = */ true,
194 [&]{
195 tatami::RetrieveCompressedSparseContentsOptions opt;
196 opt.two_pass = false;
197 opt.num_threads = options.num_threads;
198 return opt;
199 }()
200 );
201
202 // But we effectively transpose it to CSC with genes in columns.
203 const Index_ ncells = mat.ncol();
204 const auto sparse_ptr = new irlba::ParallelSparseMatrix<
205 EigenVector_,
206 EigenMatrix_,
207 I<decltype(extracted.value)>,
208 I<decltype(extracted.index)>,
209 I<decltype(extracted.pointers)>
210 >(
211 ncells,
212 ngenes,
213 std::move(extracted.value),
214 std::move(extracted.index),
215 std::move(extracted.pointers),
216 true,
217 options.num_threads
218 );
219 output.reset(sparse_ptr);
220
221 tatami::parallelize([&](const int, const Index_ start, const Index_ length) -> void {
222 const auto& pointers = sparse_ptr->get_pointers();
223 const auto& values = sparse_ptr->get_values();
224 for (Index_ g = start, end = start + length; g < end; ++g) {
225 const auto offset = pointers[g];
226 const auto next_offset = pointers[g + 1]; // increment won't overflow as 'g + 1 <= end'.
227 const Index_ num_nonzero = next_offset - offset;
228 const auto results = tatami_stats::variances::direct(values.data() + offset, num_nonzero, ncells, /* skip_nan = */ false);
229 center_v.coeffRef(g) = results.first;
230 scale_v.coeffRef(g) = results.second;
231 }
232 }, ngenes, options.num_threads);
233
234 total_var = process_scale_vector(options.scale, scale_v);
235
236 } else {
237 compute_row_means_and_variances<true>(mat, options.num_threads, center_v, scale_v);
238 total_var = process_scale_vector(options.scale, scale_v);
239
240 output.reset(
241 new irlba_tatami::Transposed<EigenVector_, EigenMatrix_, Value_, Index_, decltype(&mat)>(&mat, options.num_threads)
242 );
243 }
244
245 return prepare_deferred_matrix_for_irlba(std::move(output), options, center_v, scale_v);
246}

248template<class EigenMatrix_, typename Value_, typename Index_, class EigenVector_>
249std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > prepare_dense_matrix_for_irlba(
250 const tatami::Matrix<Value_, Index_>& mat,
251 const SimplePcaOptions<EigenVector_>& options,
252 EigenVector_& center_v,
253 EigenVector_& scale_v,
254 typename EigenVector_::Scalar& total_var
255) {
256 const Index_ ngenes = mat.nrow();
257 sanisizer::resize(center_v, ngenes);
258 sanisizer::resize(scale_v, ngenes);
259
260 if (options.realize_matrix) {
261 // Create a matrix with genes in columns.
262 const Index_ ncells = mat.ncol();
263 auto emat = std::make_unique<EigenMatrix_>(
264 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().rows())> >(ncells),
265 sanisizer::cast<I<decltype(std::declval<EigenMatrix_>().cols())> >(ngenes)
266 );
267
268 // By default, Eigen's matrices are column major. In such cases, because we want to do
269 // a transposition, we pretend it's row major during the conversion.
270 static_assert(!EigenMatrix_::IsRowMajor);
271 tatami::convert_to_dense(
272 mat,
273 /* row_major = */ true,
274 emat->data(),
275 [&]{
276 tatami::ConvertToDenseOptions opt;
277 opt.num_threads = options.num_threads;
278 return opt;
279 }()
280 );
281
282 center_v.array() = emat->array().colwise().sum();
283 if (ncells) {
284 center_v /= ncells;
285 } else {
286 std::fill(center_v.begin(), center_v.end(), std::numeric_limits<typename EigenVector_::Scalar>::quiet_NaN());
287 }
288 emat->array().rowwise() -= center_v.adjoint().array(); // applying it to avoid wasting time with deferred operations inside IRLBA.
289
290 scale_v.array() = emat->array().colwise().squaredNorm();
291 if (ncells > 1) {
292 scale_v /= ncells - 1;
293 } else {
294 std::fill(scale_v.begin(), scale_v.end(), std::numeric_limits<typename EigenVector_::Scalar>::quiet_NaN());
295 }
296
297 total_var = process_scale_vector(options.scale, scale_v);
298 if (options.scale) {
299 emat->array().rowwise() /= scale_v.adjoint().array();
300 }
301
302 return std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> >(
303 new irlba::SimpleMatrix<EigenVector_, EigenMatrix_, decltype(emat)>(std::move(emat))
304 );
305
306 } else {
307 compute_row_means_and_variances<false>(mat, options.num_threads, center_v, scale_v);
308 total_var = process_scale_vector(options.scale, scale_v);
309
310 std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > output(
311 new irlba_tatami::Transposed<EigenVector_, EigenMatrix_, Value_, Index_, decltype(&mat)>(&mat, options.num_threads)
312 );
313 return prepare_deferred_matrix_for_irlba(std::move(output), options, center_v, scale_v);
314 }

379void simple_pca_internal(
380 const tatami::Matrix<Value_, Index_>& mat,
381 const SimplePcaOptions<EigenVector_>& options,
382 SimplePcaResults<EigenMatrix_, EigenVector_>& output,
383 SubsetFunction_ subset_fun
384) {
385 irlba::EigenThreadScope t(options.num_threads);
386
387 std::unique_ptr<irlba::Matrix<EigenVector_, EigenMatrix_> > ptr;
388 if (mat.sparse()) {
389 ptr = prepare_sparse_matrix_for_irlba<EigenMatrix_>(mat, options, output.center, output.scale, output.total_variance);
390 } else {
391 ptr = prepare_dense_matrix_for_irlba<EigenMatrix_>(mat, options, output.center, output.scale, output.total_variance);
392 }
393
394 output.metrics = irlba::compute(*ptr, options.number, output.components, output.rotation, output.variance_explained, options.irlba_options);
395
396 subset_fun(output.components, output.variance_explained);
397
398 clean_up(mat.ncol(), output.components, output.variance_explained);
399 if (options.transpose) {
400 output.components.adjointInPlace();
401 }
402
403 if (!options.scale) {
404 output.scale = EigenVector_();
405 }
